Carrito Bluetooth

Un carrito de 4 motores que se maneja desde el celular. El teléfono manda letras por Bluetooth, el Arduino las lee y le dice al puente H cómo mover cada lado.

Arduino Uno Driver L298N Bluetooth HC-05 4 motores DC App: Arduino Bluetooth RC Car

Esquemático

Batería + Tierra (GND) / motores 5 V Velocidad (PWM) Dirección Datos Bluetooth

Los dos motores de cada lado van en paralelo. Todas las tierras están unidas. La línea punteada (D3 → RXD) es opcional: el carrito solo recibe datos, y si se conecta conviene bajar los 5 V a ~3.3 V con un divisor de resistencias.

Proceso

  1. Chasis y motores

    Se montaron los 4 motores en el chasis y se unieron en parejas: los dos de la izquierda a OUT1/OUT2 y los dos de la derecha a OUT3/OUT4 del L298N. Así cada lado se controla como si fuera un solo motor.

  2. Alimentación

    La batería entra al borne de 12 V del L298N. El regulador del módulo da 5 V que alimentan al Arduino, y todas las tierras (batería, L298N, Arduino y HC-05) quedan unidas para que las señales tengan la misma referencia.

  3. Señales de control

    Se quitaron los jumpers de ENA y ENB para poder regular la velocidad con PWM desde D5 y D6. Los pines IN1–IN4 van a D7–D10 y deciden hacia qué lado gira cada par de motores.

  4. Bluetooth

    El HC-05 se conectó a D2/D3 con SoftwareSerial, para dejar libre el puerto USB y poder subir el código y usar el Monitor Serie sin desconectar nada. El módulo trabaja a 9600 baud, su velocidad de fábrica.

  5. Código

    La función lado() mueve un lado con un valor de −255 a 255 (el signo es la dirección). mover() usa esa función para los dos lados, y procesar() convierte cada letra que manda la app en un movimiento.

  6. Pruebas y ajustes

    Primero se probó desde el Monitor Serie mandando letras a mano, y después desde la app. Si un lado giraba al revés, se corrige cambiando INVERTIR_IZQ o INVERTIR_DER a true en vez de volver a cablear.

Código

Ver carrito_bluetooth.ino 
// Carrito Bluetooth - Arduino Uno + L298N + HC-05
// App Android: "Arduino Bluetooth RC Car"

#include <SoftwareSerial.h>

// Bluetooth HC-05: TXD del modulo -> pin 2 del Arduino
SoftwareSerial bt(2, 3); // RX, TX

// L298N
const int ENA = 5;   // velocidad motores izquierdos
const int IN1 = 7;
const int IN2 = 8;
const int IN3 = 9;
const int IN4 = 10;
const int ENB = 6;   // velocidad motores derechos

int velocidad = 200; // 0 a 255

// Si al ir hacia adelante un lado gira al reves, cambia su valor a true
const bool INVERTIR_IZQ = false;
const bool INVERTIR_DER = false;

void lado(int en, int a, int b, int v, bool invertir) {
  if (invertir) v = -v;
  if (v > 0) {
    digitalWrite(a, HIGH);
    digitalWrite(b, LOW);
  } else if (v < 0) {
    digitalWrite(a, LOW);
    digitalWrite(b, HIGH);
  } else {
    digitalWrite(a, LOW);
    digitalWrite(b, LOW);
  }
  analogWrite(en, abs(v));
}

void mover(int izq, int der) {
  lado(ENA, IN1, IN2, izq, INVERTIR_IZQ);
  lado(ENB, IN3, IN4, der, INVERTIR_DER);
}

void procesar(char c) {
  int v = velocidad;
  switch (c) {
    case 'F': mover(v, v); break;           // adelante
    case 'B': mover(-v, -v); break;         // atras
    case 'L': mover(-v, v); break;          // girar izquierda
    case 'R': mover(v, -v); break;          // girar derecha
    case 'G': mover(v / 2, v); break;       // adelante-izquierda
    case 'I': mover(v, v / 2); break;       // adelante-derecha
    case 'H': mover(-v / 2, -v); break;     // atras-izquierda
    case 'J': mover(-v, -v / 2); break;     // atras-derecha
    case 'S': mover(0, 0); break;           // alto
    case 'q': velocidad = 255; break;       // velocidad maxima
    default:
      if (c >= '0' && c <= '9') {
        velocidad = map(c - '0', 0, 9, 90, 240);
      }
      break;
  }
}

void setup() {
  pinMode(ENA, OUTPUT);
  pinMode(ENB, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  mover(0, 0);

  Serial.begin(9600); // para probar desde el Monitor Serie
  bt.begin(9600);     // velocidad de fabrica del HC-05
  Serial.println("Carrito listo");
}

void loop() {
  if (bt.available()) {
    char c = bt.read();
    Serial.println(c);
    procesar(c);
  }
  if (Serial.available()) {
    procesar(Serial.read());
  }
}
